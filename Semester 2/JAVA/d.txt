//Read 2 matrices from the console and perform matrix addition.
import java.util.Scanner;

class Main {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int r, c;

        // Read rows and columns
        System.out.print("Enter number of rows: ");
        r = sc.nextInt();

        System.out.print("Enter number of columns: ");
        c = sc.nextInt();

        // Create matrices
        int A[][] = new int[r][c];
        int B[][] = new int[r][c];
        int sum[][] = new int[r][c];

        // Input first matrix
        System.out.println("Enter first matrix:");
        for(int i = 0; i < r; i++) {
            for(int j = 0; j < c; j++) {
                A[i][j] = sc.nextInt();
            }
        }

        // Input second matrix
        System.out.println("Enter second matrix:");
        for(int i = 0; i < r; i++) {
            for(int j = 0; j < c; j++) {
                B[i][j] = sc.nextInt();
            }
        }

        // Matrix addition
        for(int i = 0; i < r; i++) {
            for(int j = 0; j < c; j++) {
                sum[i][j] = A[i][j] + B[i][j];
            }
        }

        // Display result
        System.out.println("Sum Matrix:");

        for(int i = 0; i < r; i++) {
            for(int j = 0; j < c; j++) {
                System.out.print(sum[i][j] + " ");
            }
            System.out.println();
        }

        sc.close();
    }
}

//Read a matrix from the console and check whether it is symmetric or not.

import java.util.Scanner;

class Main {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int n;
        System.out.print("Enter size of matrix: ");
        n = sc.nextInt();

        int a[][] = new int[n][n];

        System.out.println("Enter matrix:");
        for(int i = 0; i < n; i++) {
            for(int j = 0; j < n; j++) {
                a[i][j] = sc.nextInt();
            }
        }

        boolean symmetric = true;

        for(int i = 0; i < n; i++) {
            for(int j = 0; j < n; j++) {
                if(a[i][j] != a[j][i]) {
                    symmetric = false;
                    break;
                }
            }
        }

        if(symmetric)
            System.out.println("Matrix is Symmetric");
        else
            System.out.println("Matrix is Not Symmetric");
    }
}

import java.util.Scanner;
class SkewMatrix{
	public static void main(String[] args){
		Scanner sc = new Scanner(System.in);
		
		int n,i,j;
		System.out.println("Enter the matrix size: ");
		n = sc.nextInt();
		
		int a[][] = new int[n][n];
               
		
		System.out.println("Enter the matrix: ");
		for(i=0;i<n;i++)
		{
			for(j=0;j<n;j++)
			{
			 	a[i][j] = sc.nextInt();
			 }
		}
		
		boolean skew = true;
		
		for(i=0;i<n;i++)
		{
			for(j=0;j<n;j++)
			{
				if(a[i][j] != -a[j][i])
				{
					skew = false;
					break;
				}
			}
		}
		
		if(skew)
		{
			System.out.println("The matrix is Skew-Symmetric ");
		}
		else
		{
			System.out.println("The matrix is not Skew-Symmetric ");
		}
	}
}
